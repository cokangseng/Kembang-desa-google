    <script>document.addEventListener("DOMContentLoaded", function () {

            // Helper: scroll item ke tengah container
            function centerItem(container, item) {
                if (!container || !item) return;
                const contRect = container.getBoundingClientRect();
                const itemRect = item.getBoundingClientRect();
                const current = container.scrollLeft;
                const offsetWithin = (itemRect.left - contRect.left) + current;
                const target = offsetWithin - (container.clientWidth / 2) + (item.clientWidth / 2);
                container.scrollTo({ left: Math.max(0, target), behavior: "smooth" });
            }

            /**
             * Inisialisasi auto-center untuk tab navigation SE0GAS12KG
             * @param {string} navSelector - Selector untuk <ul> nav tabs
             * @param {function} getActive - Function untuk return element li aktif
             * @param {object} observeCfg - Konfigurasi observer { targetSelector, attr }
             */
            function initAutoCenterTabs(navSelector, getActive, observeCfg) {
                document.querySelectorAll(navSelector).forEach((nav) => {
                    const activeEl = () => getActive(nav);

                    // Center saat load
                    centerItem(nav, activeEl());

                    // Fallback saat klik
                    nav.addEventListener("click", () => {
                        setTimeout(() => centerItem(nav, activeEl()), 0);
                    });

                    // MutationObserver (pantau perubahan state)
                    if (observeCfg) {
                        const mo = new MutationObserver(() => centerItem(nav, activeEl()));
                        nav.querySelectorAll(observeCfg.targetSelector).forEach((el) => {
                            mo.observe(el, { attributes: true, attributeFilter: [observeCfg.attr] });
                        });
                    }
                });
            }

            // ------------------------
            // Tabs 1: .tabbed-content .nav
            // Aktif ditandai <a role="tab" aria-selected="true">
            // ------------------------
            initAutoCenterTabs(
                ".tabbed-content .nav",
                (nav) => {
                    const a = nav.querySelector('a[role="tab"][aria-selected="true"]');
                    return a ? a.closest("li") : null;
                },
                { targetSelector: 'a[role="tab"]', attr: "aria-selected" }
            );

            // ------------------------
            // Tabs 2: .epkb-main-nav .epkb-nav-tabs
            // Aktif ditandai <li class="active">
            // ------------------------
            initAutoCenterTabs(
                ".epkb-main-nav .epkb-nav-tabs",
                (nav) => nav.querySelector("li.active"),
                { targetSelector: "li", attr: "class" }
            );

        });
